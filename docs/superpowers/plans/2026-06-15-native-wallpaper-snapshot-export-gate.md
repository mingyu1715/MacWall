# Native Wallpaper Snapshot Export Gate Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the macOS 26 native wallpaper snapshot/export gate testable without destabilizing the already-working Desktop native video path.

**Architecture:** Keep `disabled` snapshot mode as the baseline and add explicit, generated probe modes selected by `dev.sh install --snapshot-mode`. Add one-line summary logs for lifecycle requests and run one candidate at a time so crash-prone reply shapes never become the default path.

**Tech Stack:** Swift 6, CMake-generated Xcode project, ExtensionKit app extension, private `WallpaperExtensionKit`, `WallpaperAgent`, AppKit/CoreGraphics/IOSurface, Bash dev runner tests.

---

## Scope

Included:

- Generated `MacWallSnapshotProbeMode.generated.swift`
- `dev.sh install --snapshot-mode <mode>`
- source guard tests for supported modes and default `disabled`
- structured `snapshotGate` summary logs
- crash-safe candidate switch inside `MacWallSnapshotProbe`
- development-log update

Excluded:

- Main App integration
- existing `NSWindow` backend changes
- fallback PNG policy changes
- real video frame snapshot/export
- System Settings automation
- release packaging, DMG, notarization, dist artifacts

```bash
swift_case_for_snapshot_mode() {
    case "$1" in
        disabled) printf '.disabled\n' ;;
        error) printf '.error\n' ;;
        empty-object) printf '.emptyObject\n' ;;
        raw-value-retained-iosurface) printf '.rawValueRetainedIOSurface\n' ;;
        box-retained-iosurface) printf '.boxRetainedIOSurface\n' ;;
        png-data) printf '.pngData\n' ;;
        *)
            printf 'Unknown snapshot mode: %s\n' "$1" >&2
            exit 2
            ;;
    esac
}

generate_snapshot_mode_source() {
    local mode="$1"
    local swift_case
    swift_case="$(swift_case_for_snapshot_mode "$mode")"
    print_command "generate-snapshot-mode" "$SNAPSHOT_MODE_FILE" "$mode"
    if [[ "$DRY_RUN" == "1" ]]; then
        printf 'snapshot mode: %s\n' "$mode"
        return 0
    fi

    {
        printf 'enum MacWallSnapshotProbeMode: String, Sendable {\n'
        printf '    case disabled\n'
        printf '    case error\n'
        printf '    case emptyObject = "empty-object"\n'
        printf '    case rawValueRetainedIOSurface = "raw-value-retained-iosurface"\n'
        printf '    case boxRetainedIOSurface = "box-retained-iosurface"\n'
        printf '    case pngData = "png-data"\n'
        printf '}\n\n'
        printf 'enum MacWallSnapshotProbeConfiguration {\n'
        printf '    static let mode: MacWallSnapshotProbeMode = %s\n' "$swift_case"
        printf '}\n'
    } >"$SNAPSHOT_MODE_FILE"
    printf 'snapshot mode: %s\n' "$mode"
}
```

Replace the start of `cmd_install()` with option parsing:

```bash
cmd_install() {
    local snapshot_mode="$SNAPSHOT_MODE_DEFAULT"
    while [[ "$#" -gt 0 ]]; do
        case "$1" in
            --snapshot-mode)
                shift
                [[ "$#" -gt 0 ]] || {
                    printf 'install --snapshot-mode requires a mode\n' >&2
                    exit 2
                }
                snapshot_mode="$1"
                ;;
            *)
                printf 'Unknown install option: %s\n' "$1" >&2
                exit 2
                ;;
        esac
        shift
    done

    generate_snapshot_mode_source "$snapshot_mode"
```

- [ ] **Step 4: Add retained object store and minimal candidate functions**

Add these functions in `MacWallRemoteContextProbe.swift` near the existing snapshot helpers:

```swift
final class MacWallSnapshotProbeRetainedObjectStore: @unchecked Sendable {
    static let shared = MacWallSnapshotProbeRetainedObjectStore()

    private let lock = NSLock()
    private var retainedObjects: [String: [AnyObject]] = [:]

    private init() {}

    func replace(_ objects: [AnyObject], for key: String) {
        lock.lock()
        retainedObjects[key] = objects
        lock.unlock()
    }

    func clear(for key: String) {
        lock.lock()
        retainedObjects.removeValue(forKey: key)
        lock.unlock()
    }
}

private func makeSnapshotErrorResponse() -> NSError {
    NSError(
        domain: "MacWallNativeWallpaperSnapshotProbe",
        code: 2001,
        userInfo: [NSLocalizedDescriptionKey: "Explicit snapshot probe error"]
    )
}

private func makeEmptySnapshotXPC() -> AnyObject? {
    guard let snapshotXPCClass = objc_getClass("WallpaperSnapshotXPC") as? AnyClass,
          let instance = class_createInstance(snapshotXPCClass, 0) else {
        return nil
    }
    logPrivateClassLayoutOnce(snapshotXPCClass, label: "WallpaperSnapshotXPC")
    return instance as AnyObject
}

private func makeRawValueRetainedIOSurfaceSnapshot(for id: Any?) -> AnyObject? {
    guard let surface = makeSnapshotIOSurfaceForCurrentContext(id: id),
          let snapshotXPCClass = objc_getClass("WallpaperSnapshotXPC") as? AnyClass,
          let rawValueIvar = class_getInstanceVariable(snapshotXPCClass, "rawValue"),
          let instance = class_createInstance(snapshotXPCClass, 0) else {
        return nil
    }
    let key = wallpaperIDKey(from: id)
    let snapshot = instance as AnyObject
    let surfaceObject = surface as AnyObject
    MacWallSnapshotProbeRetainedObjectStore.shared.replace([surfaceObject], for: key)
    object_setIvar(snapshot, rawValueIvar, surfaceObject)
    return snapshot
}

private func makeBoxRetainedIOSurfaceSnapshot(for id: Any?) -> AnyObject? {
    guard let surface = makeSnapshotIOSurfaceForCurrentContext(id: id),
          let snapshotXPCClass = objc_getClass("WallpaperSnapshotXPC") as? AnyClass,
          let instance = class_createInstance(snapshotXPCClass, 0) else {
        return nil
    }
    let key = wallpaperIDKey(from: id)
    let snapshot = instance as AnyObject
    let pointer = Unmanaged.passUnretained(snapshot).toOpaque()
    let offset = class_getInstanceVariable(snapshotXPCClass, "box").map(ivar_getOffset) ?? 8
    let surfaceObject = surface as AnyObject
    MacWallSnapshotProbeRetainedObjectStore.shared.replace([surfaceObject], for: key)
    let surfacePointer = Unmanaged.passUnretained(surfaceObject).toOpaque()
    pointer.advanced(by: offset).storeBytes(of: surfacePointer, as: UnsafeRawPointer.self)
    return snapshot
}

private func makePNGDataSnapshot(for id: Any?) -> AnyObject? {
    guard let surface = makeSnapshotIOSurfaceForCurrentContext(id: id) else {
        return nil
    }
    return makePNGData(from: surface) as NSData?
}
```

The plan intentionally keeps `png-data` as `NSData` so the run can prove whether WallpaperAgent accepts non-`WallpaperSnapshotXPC` payloads. If XPC rejects `NSData`, classify the result and do not patch around it in the same task.
